cmake_minimum_required(VERSION 3.16)
project(multigrid_cpp VERSION 0.1.0 LANGUAGES CXX)
option(MULTIGRID_OPENMP "Use OpenMP if available" ON)
option(MULTIGRID_NATIVE "Compile for the build CPU (not portable binaries)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(multigrid_core src/field.cpp src/gauge.cpp src/clover.cpp src/dslash.cpp src/dirac.cpp src/gcr.cpp src/multigrid.cpp)
target_include_directories(multigrid_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MULTIGRID_OPENMP)
    find_package(OpenMP QUIET)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(multigrid_core PUBLIC OpenMP::OpenMP_CXX)
    endif()
endif()
if(MULTIGRID_NATIVE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(multigrid_core PUBLIC -march=native -mtune=native)
endif()
add_executable(multigrid src/main.cpp tests/validation.cpp)
target_include_directories(multigrid PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(multigrid PRIVATE multigrid_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(multigrid_core_test tests/core_test.cpp)
    target_link_libraries(multigrid_core_test PRIVATE multigrid_core)
    add_test(NAME core COMMAND multigrid_core_test)
    set_tests_properties(core PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2")
    add_test(NAME cli_help COMMAND multigrid --help)
    add_test(NAME cli_missing_value COMMAND multigrid --gauge)
    add_test(NAME cli_unknown_solver COMMAND multigrid --solver invalid)
    set_tests_properties(cli_missing_value cli_unknown_solver PROPERTIES WILL_FAIL TRUE)
endif()
