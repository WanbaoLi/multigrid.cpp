cmake_minimum_required(VERSION 3.18)
project(sun_cpu LANGUAGES CXX)
find_package(OpenMP REQUIRED)
add_executable(sun_cpu main.cpp)
target_compile_features(sun_cpu PRIVATE cxx_std_17)
target_compile_options(sun_cpu PRIVATE -O3 -march=native)
target_link_libraries(sun_cpu PRIVATE OpenMP::OpenMP_CXX)
enable_testing()
add_test(NAME sun_physics COMMAND sun_cpu --self-test)
